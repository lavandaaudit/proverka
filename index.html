// ===== TEMPORAL FLUX ANALYZER ===== const fluxCanvas = document.getElementById("fluxCanvas"); const fluxCtx = fluxCanvas.getContext("2d"); let fluxParticles = []; function resizeFlux() { fluxCanvas.width = fluxCanvas.clientWidth; fluxCanvas.height = fluxCanvas.clientHeight; fluxParticles = Array(50).fill().map(() => ({ x: Math.random()*fluxCanvas.width, y: Math.random()*fluxCanvas.height, vx: (Math.random()-0.5)*2, vy: (Math.random()-0.5)*2, size: Math.random()*3+1 })); } window.addEventListener('resize', resizeFlux); resizeFlux(); function updateFlux(gps, latency, drift) { // Новий параметр – комбінація (можна назвати flux) return ((gps/50) + (latency/250) + drift) / 3; } function drawFlux(fluxValue){ fluxCtx.fillStyle = "rgba(0,0,0,0.1)"; fluxCtx.fillRect(0,0,fluxCanvas.width,fluxCanvas.height); fluxParticles.forEach(p=>{ // рух частинок залежно від fluxValue p.x += p.vx*fluxValue*5; p.y += p.vy*fluxValue*5; if(p.x<0) p.x=fluxCanvas.width; if(p.x>fluxCanvas.width) p.x=0; if(p.y<0) p.y=fluxCanvas.height; if(p.y>fluxCanvas.height) p.y=0; fluxCtx.fillStyle = `rgba(${Math.floor(255*fluxValue)}, ${255-Math.floor(255*fluxValue)}, 150, 0.8)`; fluxCtx.beginPath(); fluxCtx.arc(p.x, p.y, p.size,0,Math.PI*2); fluxCtx.fill(); }); // Алерти const alertBox = document.getElementById("fluxAlert"); if(fluxValue<0.35){ alertBox.textContent = "🟢 STABLE FLUX · Часові потоки гармонійні"; alertBox.style.color="#00ff9c"; } else if(fluxValue<0.65){ alertBox.textContent = "🟡 MODERATE FLUX · Легкі коливання"; alertBox.style.color="#ffaa00"; } else{ alertBox.textContent = "🔴 HIGH FLUX · Потік нестабільний"; alertBox.style.color="#ff3366"; } document.getElementById("fluxValue").textContent = fluxValue.toFixed(3); requestAnimationFrame(()=>drawFlux(fluxValue)); } // ===== ІНТЕГРАЦІЯ У ГОЛОВНИЙ ЦИКЛ ===== async function mainLoop(){ updateAtomicTime(); const gps = updateGpsOffset(); const latency = await measureLatency(); const drift = calcDrift(gps, latency); document.getElementById("drift").textContent = drift.toFixed(3); const index = calcTimeIbonariumIndex(gps, latency, drift); document.getElementById("timeIndex").textContent = index.toFixed(3); updateAlert(index); updateMatrixFromIndex(index, drift); // Flux const flux = updateFlux(gps, latency, drift); drawFlux(flux); } setInterval(mainLoop, 500); mainLoop();
